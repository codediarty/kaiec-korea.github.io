// 사이트는 POST(폼 형식)로 보내고, 실패하면 GET(주소에 데이터)으로 다시 보냅니다. 파라미터가 없으면 상태 확인 문구만 보여줍니다.
function doGet(e) {
  var p = (e && e.parameter) || {};
  if (p.name || p.email) {
    try { return out_(handle_(p)); }
    catch (err) { logError_(err, JSON.stringify(p)); return out_('error: ' + err); }
  }
  return out_('KAIEC 접수 수신 대기 중 (ok) · v' + VERSION);
}

function doPost(e) {
  var raw = '';
  try {
    var d = {};
    raw = e && e.postData ? String(e.postData.contents || '') : '';
    try { d = JSON.parse(raw); } catch (err) { d = (e && e.parameter) || {}; }
    if (!d || typeof d !== 'object') d = {};
    return out_(handle_(d));
  } catch (err2) {
    logError_(err2, raw);
    return out_('error: ' + err2);
  }
}

function out_(s) { return ContentService.createTextOutput(String(s)); }

function handle_(d) {
  var isJoin = String(d.type || '') === 'join' || /^\[KAIEC 참여\]/.test(String(d.course || ''));
  return isJoin ? saveJoin_(d) : saveApply_(d);
}

function saveApply_(d) {
  var ss = getSs_();
  var sh = tab_(ss, APPLY_SHEET, APPLY_COLS);
  var c = applyCols_(sh);
  if (isDupApply_(sh, c, d)) return 'duplicate';
  var row = [];
  for (var i = 0; i < c.n; i++) row.push('');
  var when = new Date();
  row[c.date] = when; row[c.course] = str_(d.course); row[c.name] = str_(d.name);
  row[c.phone] = str_(d.phone); row[c.email] = str_(d.email); row[c.job] = str_(d.job); row[c.purpose] = str_(d.purpose);
  sh.appendRow(row);
  var course = str_(d.course) || 'AI윤리전문가 양성과정';
  var rows = [['과정', course], ['성명', str_(d.name)], ['이메일', str_(d.email)], ['휴대전화', str_(d.phone) || '(미기재)'],
              ['직업 · 활동 분야', str_(d.job)], ['활용 목적', str_(d.purpose) || '(선택 안 함)'], ['접수 일시', stamp_(when)]];
  var m = adminMail_(course + ' 신청이 접수되었습니다', rows,

    [['접수 시트 열기', sheetUrl_(ss, sh)]]);
  notifyAdmin_('[AI윤리전문가 양성과정 신청] ' + (str_(d.name) || '(성명 없음)'), m, d.email);

}

function applyCols_(sh) {
  var head = headRow_(sh);
  var pi = head.indexOf('활용 목적');
  if (pi < 0) pi = head.indexOf('자격 취득 목적');   // 4차 시트의 옛 열 제목도 그대로 사용
  if (pi < 0) { pi = head.length; sh.getRange(1, pi + 1).setValue('활용 목적'); head.push('활용 목적'); }
  function idx(name, fallback) { var i = head.indexOf(name); return i >= 0 ? i : fallback; }
  return { n: head.length, date: idx('접수일시', 0), course: idx('응시 과정', 1), name: idx('성명', 2),
           phone: idx('휴대전화', 3), email: idx('이메일', 4), job: idx('직업/활동 분야', 5), purpose: pi };
}

// 같은 사람이 2분 안에 두 번 들어오면(새로고침 등) 한 번만 기록
function isDupApply_(sh, c, d) {
  var last = sh.getLastRow();
  if (last < 2) return false;
  var n = Math.min(20, last - 1);
  var rows = sh.getRange(last - n + 1, 1, n, c.n).getValues();
  var now = new Date().getTime();
  for (var i = 0; i < rows.length; i++) {
    var t = rows[i][c.date] instanceof Date ? rows[i][c.date].getTime() : 0;
    if (str_(rows[i][c.name]) === str_(d.name) && lower_(rows[i][c.email]) === lower_(d.email) && now - t < 120000) return true;
  }
  return false;
}

/* ═════════════════════════ 위원 참여 신청 (/join/) ═════════════════════════ */

function saveJoin_(d) {
  var ss = getSs_();
  var sh = tab_(ss, JOIN_SHEET, JOIN_COLS);
  var c = joinCols_(sh);
  var kind = str_(d.course).replace(/^\[KAIEC 참여\]\s*/, '') || '미지정';
  var email = lower_(d.email);
  if (isDupJoin_(sh, c, email, kind)) return 'duplicate';

  var when = new Date();
  var row = [];
  for (var i = 0; i < c.n; i++) row.push('');
  row[c.date] = when; row[c.kind] = kind; row[c.name] = str_(d.name); row[c.email] = str_(d.email);
  row[c.phone] = str_(d.phone); row[c.job] = str_(d.job); row[c.org] = str_(d.org);
  row[c.motive] = str_(d.motive); row[c.purpose] = str_(d.purpose);
  sh.appendRow(row);   // '지원자 메일' · '비고' 칸은 비워 둡니다(직접 연락한 뒤 메모용)

  var rows = [['참여 구분', kind], ['성명', str_(d.name)], ['이메일', str_(d.email)], ['휴대전화', str_(d.phone) || '(미기재)'],
              ['직업 · 활동 분야', str_(d.job)], ['소속', str_(d.org) || '(미기재)'], ['기대하는 것', str_(d.motive)],
              ['자기소개', str_(d.purpose) || '(없음)'], ['접수 일시', stamp_(when)]];
  var next = isCampaign_(kind)
    ? '지원자에게 자동 메일은 가지 않았습니다. 검토 후 지원자에게 캠페인위원 참여 제안(활동 안내서)과 활동 안내를 직접 보내 주십시오. 이 메일에 답장하면 지원자에게 바로 갑니다.'
    : '지원자에게 자동 메일은 가지 않았습니다. 검토 후 3~5일 안에 지원자에게 결과와 활동 안내를 보내 주십시오(위촉 시 홈페이지 공식 위원 명단에 등록). 이 메일에 답장하면 지원자에게 바로 갑니다.';
  var m = adminMail_('위원 참여 신청이 접수되었습니다', rows, next, [['위원 신청 탭 열기', sheetUrl_(ss, sh)]]);
  notifyAdmin_('[한국AI윤리위원회(KAIEC) ' + kind + ' 지원] ' + (str_(d.name) || '(성명 없음)'), m, d.email);
  return 'ok';
}

function joinCols_(sh) {
  var head = headRow_(sh);
  // 옛 4차 탭(9열)이면 빠진 열 제목을 뒤에 붙임
  JOIN_COLS.forEach(function (h) {
    if (head.indexOf(h) < 0) { sh.getRange(1, head.length + 1).setValue(h); head.push(h); }
  });
  function idx(name) { return head.indexOf(name); }
  return { n: head.length, date: idx('접수일시'), kind: idx('참여 구분'), name: idx('성명'), email: idx('이메일'), phone: idx('휴대전화'),
           job: idx('직업/활동 분야'), org: idx('소속'), motive: idx('기대하는 것'), purpose: idx('지원 사유·자기소개') };
}

function isDupJoin_(sh, c, email, kind) {
  var last = sh.getLastRow();
  if (last < 2 || !email) return false;
  var n = Math.min(20, last - 1);
  var rows = sh.getRange(last - n + 1, 1, n, c.n).getValues();
  var now = new Date().getTime();
  for (var i = 0; i < rows.length; i++) {
    var t = rows[i][c.date] instanceof Date ? rows[i][c.date].getTime() : 0;
    if (lower_(rows[i][c.email]) === email && str_(rows[i][c.kind]) === kind && now - t < 120000) return true;
  }
  return false;
}

function isCampaign_(kind) { return /캠페인/.test(String(kind || '')); }

// 관리자 알림(HTML 표 + 텍스트)
function adminMail_(title, rows, next, links) {
  var text = [title, ''];
  rows.forEach(function (r) { text.push(r[0] + ' : ' + r[1]); });
  if (next) text.push('', next);
  (links || []).forEach(function (l) { text.push('', l[0] + ': ' + l[1]); });
  var tr = rows.map(function (r, i) {
    var b = i < rows.length - 1 ? 'border-bottom:1px solid #E6EAF0;' : '';
    return '<tr><th align="left" style="' + b + 'padding:9px 12px;background:#F4F6FA;font-weight:600;color:#44546B;white-space:nowrap;vertical-align:top;width:110px">' + esc_(r[0]) + '</th>' +
           '<td style="' + b + 'padding:9px 12px;color:#1F2937;vertical-align:top;word-break:break-all">' + esc_(r[1]).replace(/\n/g, '<br>') + '</td></tr>';
  }).join('');
  var html =
    '<div style="font-family:' + FONT + ';font-size:14.5px;line-height:1.7;color:#1F2937;max-width:640px;margin:0 auto;padding:4px 2px;word-break:keep-all">' +
      '<table role="presentation" cellpadding="0" cellspacing="0" width="100%" style="border-collapse:collapse;margin:0 0 22px"><tr>' +
        '<td style="border-top:3px solid #0B2A4A;border-bottom:1px solid #E6EAF0;padding:13px 0 11px;font-size:15px;font-weight:800;color:#0B2A4A;letter-spacing:-0.3px">한국AI윤리위원회</td>' +
        '<td align="right" style="border-top:3px solid #0B2A4A;border-bottom:1px solid #E6EAF0;padding:13px 0 11px 12px;font-size:11px;font-weight:700;color:#8A94A6;letter-spacing:2.5px;white-space:nowrap">KAIEC · 접수 알림</td>' +
      '</tr></table>' +
      '<p style="margin:0 0 16px;font-size:18px;font-weight:800;color:#0B2A4A;letter-spacing:-0.3px">' + esc_(title) + '</p>' +
      '<table cellpadding="0" cellspacing="0" style="width:100%;border-collapse:collapse;border:1px solid #E6EAF0;font-size:14px;margin:0 0 16px">' + tr + '</table>' +
      (next ? '<p style="margin:0 0 16px;padding:12px 14px;background:#F8FAFC;border-left:3px solid #0B2A4A;color:#374151">' + esc_(next) + '</p>' : '') +
      (links || []).map(function (l) { return '<p style="margin:0 0 8px"><a href="' + esc_(l[1]) + '" style="color:#0B5CAB;font-weight:700">' + esc_(l[0]) + '</a></p>'; }).join('') +
      '<p style="margin:18px 0 0;font-size:12px;color:#9AA3B2">kaiec.kr 접수 시트 연동 v' + esc_(VERSION) + ' · 이 메일은 관리자에게만 발송됩니다.</p>' +
    '</div>';
  return { text: text.join('\n'), html: html };
}

// 관리자에게 알림. 지원자 이메일이 올바르면 '답장' 받는 곳을 지원자로 둡니다(답장하면 지원자에게 감)
function notifyAdmin_(subject, m, applicantEmail) {
  if (!NOTIFY) return;
  var o = { to: NOTIFY, subject: subject, body: m.text, htmlBody: m.html, name: 'KAIEC 접수 알림' };
  var rt = str_(applicantEmail);
  if (/^[^@\s,;<>]+@[^@\s,;<>]+\.[^@\s,;<>]+$/.test(rt)) o.replyTo = rt;
  try { MailApp.sendEmail(o); }
  catch (e) {
    logError_(e, '관리자 알림 ' + subject);
    if (o.replyTo) { try { delete o.replyTo; MailApp.sendEmail(o); } catch (e2) { logError_(e2, '관리자 알림 재시도 ' + subject); } }
  }
}

function tab_(ss, name, cols) {
  var sh = ss.getSheetByName(name) || ss.insertSheet(name);

    sh.appendRow(cols);
    sh.setFrozenRows(1);

  }

function headRow_(sh) {
  var w = Math.max(1, sh.getLastColumn());
  return sh.getRange(1, 1, 1, w).getValues()[0].map(function (v) { return String(v == null ? '' : v).trim(); }).filter(function (v, i, arr) {
    // 뒤쪽의 빈 제목은 버림(가운데 빈 칸은 자리 유지)
    for (var j = i; j < arr.length; j++) if (arr[j]) return true;
    return false;
  });
}
