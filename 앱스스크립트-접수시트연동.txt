 KAIEC AI윤리전문가 응시 접수 → 구글 스프레드시트 자동 수집

[1] Apps Script 편집기에서 기존 코드를 전부 지우고 아래 코드를 통째로
    붙여넣은 뒤 저장(Ctrl+S)합니다.

──────────────── 여기부터 복사 ────────────────

var SHEET_ID = '1oahWRauDAMnUkySjrl1MBhJQDR1ThOYpAXBFOldrUSA';   // 접수 스프레드시트 ID
var NOTIFY   = 'kaiec.korea@gmail.com, skkc.office@gmail.com';   // 접수 알림 메일 (쉼표로 추가 가능)

function getSheet_() {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sh = ss.getSheetByName('접수') || ss.insertSheet('접수');
  if (sh.getLastRow() === 0) {
    sh.appendRow(['접수일시', '응시 과정', '성명', '휴대전화', '이메일', '직업/활동 분야']);
    sh.setFrozenRows(1);
  }
  return sh;
}

// 같은 사람이 2분 안에 두 번 들어오면(새로고침 등) 한 번만 기록
function isDuplicate_(sh, d) {
  var last = sh.getLastRow();
  if (last < 2) return false;
  var n = Math.min(20, last - 1);
  var rows = sh.getRange(last - n + 1, 1, n, 6).getValues();
  var now = new Date().getTime();
  for (var i = 0; i < rows.length; i++) {
    var t = rows[i][0] instanceof Date ? rows[i][0].getTime() : 0;
    if (rows[i][2] === (d.name || '') && rows[i][4] === (d.email || '') && now - t < 120000) return true;
  }
  return false;
}

function saveAndNotify_(d) {
  var sh = getSheet_();
  if (isDuplicate_(sh, d)) return 'duplicate';
  sh.appendRow([new Date(), d.course || '', d.name || '', d.phone || '', d.email || '', d.job || '']);
  MailApp.sendEmail(NOTIFY,
    '[AI윤리전문가 1기 응시] ' + (d.name || '') + ' · ' + (d.course || ''),
    '새 응시 접수가 등록되었습니다.\n\n' +
    '응시 과정 : ' + (d.course || '') + '\n성명 : ' + (d.name || '') +
    '\n휴대전화 : ' + (d.phone || '') + '\n이메일 : ' + (d.email || '') +
    '\n직업/활동 분야 : ' + (d.job || '') +
    '\n\n스프레드시트: https://docs.google.com/spreadsheets/d/' + SHEET_ID);
  return 'ok';
}

function logError_(err, raw) {
  try {
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var log = ss.getSheetByName('오류로그') || ss.insertSheet('오류로그');
    log.appendRow([new Date(), String(err), raw || '']);
  } catch (e2) {}
}

// 사이트는 GET(주소에 데이터)으로 보냅니다. 파라미터가 없으면 상태 확인 문구만 보여줍니다.
function doGet(e) {
  var p = (e && e.parameter) || {};
  if (p.name || p.email) {
    try { return ContentService.createTextOutput(saveAndNotify_(p)); }
    catch (err) { logError_(err, JSON.stringify(p)); return ContentService.createTextOutput('error: ' + err); }
  }
  return ContentService.createTextOutput('KAIEC 접수 수신 대기 중 (ok)');
}

// 예전 방식(POST)으로 들어오는 경우도 그대로 받습니다.
function doPost(e) {
  try {
    var d = {};
    try { d = JSON.parse(e.postData.contents); } catch (err) { d = (e && e.parameter) || {}; }
    return ContentService.createTextOutput(saveAndNotify_(d));
  } catch (err) {
    logError_(err, e && e.postData ? e.postData.contents : '');
    return ContentService.createTextOutput('error: ' + err);
  }
}

// 편집기에서 실행하면 테스트 줄 1개 + 알림 메일 (권한 승인용)

                  email: 'test@kaiec.kr', job: '테스트'});
}
