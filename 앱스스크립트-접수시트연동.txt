 KAIEC AI윤리전문가 응시 접수 → 구글 스프레드시트 자동 수집
 (4차 개정: GET·POST 모두 수신 + 상태 확인 페이지 + 중복 방지 + 오류 기록
  + 시트 열 구성 자동 감지 + '자격 취득 목적' 열 자동 추가, 2026.09.12)

 ※ 이 코드로 한 번 다시 올리면(붙여넣기 → 저장 → 배포 관리 → 새 버전) 두 가지가 추가됩니다.
   1) 응시 접수 폼의 '자격 취득 및 활용 목적' 답변이 '접수' 탭 마지막 열에 저장됩니다.
   2) KAIEC 참여 신청(kaiec.kr/join/)이 '참여신청' 탭에 따로 쌓이고 "[KAIEC 참여 신청]" 제목의 알림 메일이 옵니다.
   다시 올리기 전에도 참여 신청은 '접수' 탭에 "[KAIEC 참여] 운영위원"처럼 구분 표시가 붙어 기록되고
   알림 메일도 오므로 접수 자체는 누락되지 않습니다.

var SHEET_ID = '1H-R2a6Q1fmoetA7hFo6KrF2NkzSWC9ugiFSUoKkdHcs';   // 접수 스프레드시트 ID (스크립트가 연결된 시트가 있으면 그 시트를 우선 사용)
var NOTIFY   = 'kaiec.korea@gmail.com, skkc.office@gmail.com';   // 접수 알림 메일 (쉼표로 추가 가능)

function getSs_() {
  var ss = null;
  try { ss = SpreadsheetApp.getActiveSpreadsheet(); } catch (e) {}
  return ss || SpreadsheetApp.openById(SHEET_ID);
}

function getSheet_() {
  var ss = getSs_();
  var sh = ss.getSheetByName('접수') || ss.insertSheet('접수');
  if (sh.getLastRow() === 0) {
    sh.appendRow(['접수일시', '응시 과정', '성명', '휴대전화', '이메일', '직업/활동 분야', '자격 취득 목적']);
    sh.setFrozenRows(1);
  }
  return sh;
}

// 1행 제목을 읽어 열 위치를 찾습니다 (예전 8열 시트든 6열 시트든 그대로 동작).
// '자격 취득 목적' 열이 없으면 맨 뒤에 자동으로 추가합니다 (기존 시트도 그대로 사용 가능).
function cols_(sh) {
  var head = sh.getRange(1, 1, 1, Math.max(6, sh.getLastColumn())).getValues()[0];
  var pi = head.indexOf('자격 취득 목적');
  if (pi < 0) { pi = sh.getLastColumn(); sh.getRange(1, pi + 1).setValue('자격 취득 목적'); head.push('자격 취득 목적'); }
  function idx(name, fallback) { var i = head.indexOf(name); return i >= 0 ? i : fallback; }
  return { n: head.length, date: idx('접수일시', 0), course: idx('응시 과정', 1), name: idx('성명', 2),
           phone: idx('휴대전화', 3), email: idx('이메일', 4), job: idx('직업/활동 분야', 5), purpose: pi };
}

// 같은 사람이 2분 안에 두 번 들어오면(새로고침 등) 한 번만 기록
function isDuplicate_(sh, c, d) {
  var last = sh.getLastRow();
  if (last < 2) return false;
  var n = Math.min(20, last - 1);
  var rows = sh.getRange(last - n + 1, 1, n, c.n).getValues();
  var now = new Date().getTime();
  for (var i = 0; i < rows.length; i++) {
    var t = rows[i][c.date] instanceof Date ? rows[i][c.date].getTime() : 0;
    if (rows[i][c.name] === (d.name || '') && rows[i][c.email] === (d.email || '') && now - t < 120000) return true;
  }
  return false;
}

function saveAndNotify_(d) {
  var sh = getSheet_();
  var c = cols_(sh);
  if (isDuplicate_(sh, c, d)) return 'duplicate';
  var row = [];
  for (var i = 0; i < c.n; i++) row.push('');
  row[c.date] = new Date(); row[c.course] = d.course || ''; row[c.name] = d.name || '';
  row[c.phone] = d.phone || ''; row[c.email] = d.email || ''; row[c.job] = d.job || '';
  row[c.purpose] = d.purpose || '';
  sh.appendRow(row);
  MailApp.sendEmail(NOTIFY,
    '[AI윤리전문가 1기 응시] ' + (d.name || '') + ' · ' + (d.course || ''),
    '새 응시 접수가 등록되었습니다.\n\n' +
    '응시 과정 : ' + (d.course || '') + '\n성명 : ' + (d.name || '') +
    '\n휴대전화 : ' + (d.phone || '') + '\n이메일 : ' + (d.email || '') +
    '\n직업/활동 분야 : ' + (d.job || '') +
    '\n자격 취득 목적 : ' + (d.purpose || '') +
    '\n\n스프레드시트: https://docs.google.com/spreadsheets/d/' + getSs_().getId());
  return 'ok';
}

// KAIEC 참여 신청(kaiec.kr/join/)은 '참여신청' 탭에 따로 기록하고 알림 메일을 보냅니다. (type=join 으로 들어옴)
function saveJoin_(d) {
  var ss = getSs_();
  var sh = ss.getSheetByName('참여신청') || ss.insertSheet('참여신청');
  if (sh.getLastRow() === 0) {
    sh.appendRow(['접수일시', '참여 구분', '성명', '이메일', '휴대전화', '소속/직업', '참여 동기·희망 활동']);
    sh.setFrozenRows(1);
  }
  var kind = String(d.course || '').replace('[KAIEC 참여] ', '');
  sh.appendRow([new Date(), kind, d.name || '', d.email || '', d.phone || '', d.job || '', d.purpose || '']);
  MailApp.sendEmail(NOTIFY,
    '[KAIEC 참여 신청] ' + (d.name || '') + ' · ' + kind,
    '새 참여 신청이 접수되었습니다.\n\n' +
    '참여 구분 : ' + kind + '\n성명 : ' + (d.name || '') + '\n이메일 : ' + (d.email || '') +
    '\n휴대전화 : ' + (d.phone || '') + '\n소속/직업 : ' + (d.job || '') +
    '\n참여 동기·희망 활동 : ' + (d.purpose || '') +
    '\n\n스프레드시트(참여신청 탭): https://docs.google.com/spreadsheets/d/' + ss.getId());
  return 'ok';
}

function handle_(d) {
  return (d.type === 'join') ? saveJoin_(d) : saveAndNotify_(d);
}

function logError_(err, raw) {
  try {
    var log = getSs_().getSheetByName('오류로그') || getSs_().insertSheet('오류로그');
    log.appendRow([new Date(), String(err), raw || '']);
  } catch (e2) {}
}

// 사이트는 GET(주소에 데이터)으로 보냅니다. 파라미터가 없으면 상태 확인 문구만 보여줍니다.
function doGet(e) {
  var p = (e && e.parameter) || {};
  if (p.name || p.email) {
    try { return ContentService.createTextOutput(handle_(p)); }
    catch (err) { logError_(err, JSON.stringify(p)); return ContentService.createTextOutput('error: ' + err); }
  }
  return ContentService.createTextOutput('KAIEC 접수 수신 대기 중 (ok)');
}

// 예전 방식(POST)으로 들어오는 경우도 그대로 받습니다.
function doPost(e) {
  try {
    var d = {};
    try { d = JSON.parse(e.postData.contents); } catch (err) { d = (e && e.parameter) || {}; }
    return ContentService.createTextOutput(handle_(d));
  } catch (err) {
    logError_(err, e && e.postData ? e.postData.contents : '');
    return ContentService.createTextOutput('error: ' + err);
  }
}

// 편집기에서 한 번만 실행: 예전 시트(OLD_ID)의 '접수' 탭 기록을 이 시트로 복사합니다 (이미 있는 줄은 건너뜀)
function migrateOld() {
  var src = SpreadsheetApp.openById(OLD_ID).getSheetByName('접수');
  if (!src || src.getLastRow() < 2) { Logger.log('옮길 기록 없음'); return; }
  var sv = src.getDataRange().getValues();
  var sh = getSheet_();
  var c = cols_(sh);
  var existing = {};
  var tv = sh.getDataRange().getValues();
  for (var i = 1; i < tv.length; i++) {
    var t = tv[i][c.date] instanceof Date ? tv[i][c.date].getTime() : String(tv[i][c.date]);
    existing[tv[i][c.name] + '|' + tv[i][c.email] + '|' + t] = true;
  }
  var head = sv[0];
  function col(name) { return head.indexOf(name); }
  var moved = 0;
  for (var r = 1; r < sv.length; r++) {
    var row = sv[r];
    var when = row[col('접수일시')];
    var key = row[col('성명')] + '|' + row[col('이메일')] + '|' + (when instanceof Date ? when.getTime() : String(when));
    if (existing[key] || !row[col('성명')]) continue;
    appendRow_(sh, c, when, { course: row[col('응시 과정')], name: row[col('성명')], phone: row[col('휴대전화')],
                              email: row[col('이메일')], job: row[col('직업/활동 분야')] });
    existing[key] = true; moved++;
  }
  Logger.log('옮긴 줄: ' + moved);
}
