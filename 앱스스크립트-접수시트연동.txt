// 사이트는 POST(폼 형식)로 보내고, 실패하면 GET(주소에 데이터)으로 다시 보냅니다. 파라미터가 없으면 상태 확인 문구만 보여줍니다.
function doGet(e) {
  var p = (e && e.parameter) || {};
  if (p.name || p.email) {
    try { return out_(handle_(p)); }
    catch (err) { logError_(err, JSON.stringify(p)); return out_('error: ' + err); }
  }
  return out_('KAIEC 접수 수신 대기 중 (ok) · v' + VERSION);
}

function doPost(e) {
  var raw = '';
  try {
    var d = {};
    raw = e && e.postData ? String(e.postData.contents || '') : '';
    try { d = JSON.parse(raw); } catch (err) { d = (e && e.parameter) || {}; }
    if (!d || typeof d !== 'object') d = {};
    return out_(handle_(d));
  } catch (err2) {
    logError_(err2, raw);
    return out_('error: ' + err2);
  }
}

function out_(s) { return ContentService.createTextOutput(String(s)); }

function handle_(d) {
  var isJoin = String(d.type || '') === 'join' || /^\[KAIEC 참여\]/.test(String(d.course || ''));
  return isJoin ? saveJoin_(d) : saveApply_(d);
}

function applyCols_(sh) {
  var head = headRow_(sh);
  var pi = head.indexOf('활용 목적');
  if (pi < 0) pi = head.indexOf('자격 취득 목적');   // 4차 시트의 옛 열 제목도 그대로 사용
  if (pi < 0) { pi = head.length; sh.getRange(1, pi + 1).setValue('활용 목적'); head.push('활용 목적'); }
  function idx(name, fallback) { var i = head.indexOf(name); return i >= 0 ? i : fallback; }
  return { n: head.length, date: idx('접수일시', 0), course: idx('응시 과정', 1), name: idx('성명', 2),
           phone: idx('휴대전화', 3), email: idx('이메일', 4), job: idx('직업/활동 분야', 5), purpose: pi };
}

// 같은 사람이 2분 안에 두 번 들어오면(새로고침 등) 한 번만 기록
function isDupApply_(sh, c, d) {
  var last = sh.getLastRow();
  if (last < 2) return false;
  var n = Math.min(20, last - 1);
  var rows = sh.getRange(last - n + 1, 1, n, c.n).getValues();
  var now = new Date().getTime();
  for (var i = 0; i < rows.length; i++) {
    var t = rows[i][c.date] instanceof Date ? rows[i][c.date].getTime() : 0;
    if (str_(rows[i][c.name]) === str_(d.name) && lower_(rows[i][c.email]) === lower_(d.email) && now - t < 120000) return true;
  }
  return false;
}

function saveJoin_(d) {

  var sh = tab_(ss, JOIN_SHEET, JOIN_COLS);

  var name = cleanName_(d.name);
  var email = lower_(d.email);
  if (isDupJoin_(sh, c, email, kind)) return 'duplicate';

  // 1) 시트 기록
  var when = new Date();
  var row = [];
  for (var i = 0; i < c.n; i++) row.push('');

  var rows = [['참여 구분', kind], ['성명', str_(d.name)], ['이메일', str_(d.email)], ['휴대전화', str_(d.phone) || '(미기재)'],
              ['직업 · 활동 분야', str_(d.job)], ['소속', str_(d.org) || '(미기재)'], ['기대하는 것', str_(d.motive)],
              ['자기소개', str_(d.purpose) || '(없음)'], ['접수 일시', stamp_(when)], ['지원자 자동 메일', mailResult || '(보내지 않음)']];
  var proposed = mailResult.indexOf(PROPOSAL_MARK) === 0;

function joinCols_(sh) {
  var head = headRow_(sh);
  // 옛 4차 탭(9열)이면 빠진 열 제목을 뒤에 붙임
  JOIN_COLS.forEach(function (h) {
    if (head.indexOf(h) < 0) { sh.getRange(1, head.length + 1).setValue(h); head.push(h); }
  });
  function idx(name) { return head.indexOf(name); }
  return { n: head.length, date: idx('접수일시'), kind: idx('참여 구분'), name: idx('성명'), email: idx('이메일'), phone: idx('휴대전화'),
           job: idx('직업/활동 분야'), org: idx('소속'), motive: idx('기대하는 것'), purpose: idx('지원 사유·자기소개'),
           mail: idx('지원자 메일'), note: idx('비고') };
}

function isDupJoin_(sh, c, email, kind) {
  var last = sh.getLastRow();
  if (last < 2 || !email) return false;
  var n = Math.min(20, last - 1);
  var rows = sh.getRange(last - n + 1, 1, n, c.n).getValues();
  var now = new Date().getTime();
  for (var i = 0; i < rows.length; i++) {
    var t = rows[i][c.date] instanceof Date ? rows[i][c.date].getTime() : 0;
    if (lower_(rows[i][c.email]) === email && str_(rows[i][c.kind]) === kind && now - t < 120000) return true;
  }
  return false;
}

function isCampaign_(kind) { return /캠페인/.test(String(kind || '')); }

// 같은 이메일로 제안 메일을 이미 보냈으면 true
function alreadyProposed_(sh, c, email) {
  var last = sh.getLastRow();
  if (last < 2 || !email) return false;
  var rows = sh.getRange(2, 1, last - 1, c.n).getValues();
  for (var i = 0; i < rows.length; i++) {
    if (lower_(rows[i][c.email]) === email && String(rows[i][c.mail] || '').indexOf(PROPOSAL_MARK) === 0) return true;
  }
  return false;
}

// 지원자 자동 메일. 결과 문구를 돌려주고 '지원자 메일' 칸에 기록됩니다
function sendApplicant_(sh, c, d, kind, name, email) {
  if (!MAIL.APPLICANT_ON) return '보내지 않음(자동 메일 꺼짐)';
  if (!email || email.indexOf('@') < 1 || !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) return '건너뜀: 이메일 형식 오류';
  if (MAIL.SKIP_EMAIL && new RegExp(MAIL.SKIP_EMAIL, 'i').test(email)) return '건너뜀: 테스트 · 내부 주소';
  var quota = 0;
  try { quota = MailApp.getRemainingDailyQuota(); } catch (e) { quota = 0; }
  if (quota <= MAIL.QUOTA_RESERVE) return '보류: 오늘 발송 한도 부족(남은 ' + quota + '통). 내일 previewMails 확인 후 직접 발송';
  if (isCampaign_(kind) && MAIL.CAMPAIGN_PROPOSAL) {
    if (alreadyProposed_(sh, c, email)) return '건너뜀: 같은 이메일에 제안 메일을 이미 보냄';
    var g = guideState_(false);
    var m = campaignMail_(name);
    var extra = {};
    if (g.ok) extra.attachments = [g.file.getBlob().setName(MAIL.ATTACH_NAME)];
    sendMail_(email, m.subject, m.text, m.html, extra);
    return PROPOSAL_MARK + ' ' + stamp_(new Date()) + (g.ok ? ' · 안내서 첨부' : ' · 안내서 미첨부(' + g.msg + ')');
  }
  var a = ackMail_(name, kind);
  sendMail_(email, a.subject, a.text, a.html, {});
  return '접수 확인 메일 발송 ' + stamp_(new Date());
}

function senderPick_() {
  if (SENDER_PICK_) return SENDER_PICK_;
  var me = '', aliases = [];
  try { me = lower_(Session.getEffectiveUser().getEmail()); } catch (e) { me = ''; }
  try { aliases = GmailApp.getAliases().map(function (a) { return lower_(a); }); } catch (e2) { aliases = []; }
  var cands = [MAIL.FROM, MAIL.FROM_FALLBACK].map(lower_).filter(function (x) { return !!x; });
  var pick = '';
  for (var i = 0; i < cands.length; i++) { if (cands[i] === me || aliases.indexOf(cands[i]) >= 0) { pick = cands[i]; break; } }
  var reply = lower_(MAIL.REPLY_TO) || pick || cands[cands.length - 1] || me;
  SENDER_PICK_ = { from: pick, reply: reply, me: me, aliases: aliases, official: !!pick && pick === cands[0] };
  return SENDER_PICK_;
}

  }

// fresh = true 면 저장해 둔 파일 ID 를 쓰지 않고 드라이브를 다시 찾습니다(checkSetup)
function guideState_(fresh) {
  var props = PropertiesService.getScriptProperties();
  var f = null;
  if (MAIL.GUIDE_FILE_ID) {
    f = openFile_(MAIL.GUIDE_FILE_ID);
    if (!f) return { ok: false, msg: 'MAIL.GUIDE_FILE_ID(' + MAIL.GUIDE_FILE_ID + ') 파일을 열 수 없음' };
  }
  if (!f && !fresh) {
    var saved = props.getProperty('GUIDE_ID');
    if (saved) f = openFile_(saved);
  }
  if (!f) {
    f = searchGuide_();
    if (f) props.setProperty('GUIDE_ID', f.getId()); else props.deleteProperty('GUIDE_ID');
  }
  if (!f) return { ok: false, msg: "드라이브에 이름에 '" + MAIL.GUIDE_KEYWORD + "' 가 들어간 PDF 없음" };
  var size = 0;
  try { size = f.getSize(); } catch (e) { size = 0; }
  if (size > 20 * 1024 * 1024) return { ok: false, msg: '안내서 PDF 가 20MB 초과(' + kb_(size) + ')' };
  return { ok: true, file: f, name: f.getName(), size: size };
}

function openFile_(id) {
  try { var f = DriveApp.getFileById(id); return f.isTrashed() ? null : f; } catch (e) { return null; }
}

function searchGuide_() {
  var kw = String(MAIL.GUIDE_KEYWORD || '캠페인위원');
  var q = kw.replace(/\\/g, '\\\\').replace(/'/g, "\\'");
  var best = null, bestScore = -1, seen = 0;
  var consider = function (it, cap) {
    while (it.hasNext() && seen < cap) {
      var f = it.next(); seen++;
      try {
        if (f.isTrashed() || f.getMimeType() !== 'application/pdf') continue;
        var nm = f.getName();
        if (nm.indexOf(kw) < 0) continue;
        var score = (nm.indexOf('안내서') >= 0 ? 4 : 0) + (nm.indexOf('활동') >= 0 ? 2 : 0) + (nm.indexOf('안내') >= 0 ? 1 : 0);
        if (score > bestScore || (score === bestScore && f.getLastUpdated() > best.getLastUpdated())) { best = f; bestScore = score; }
      } catch (e) { /* 읽을 수 없는 파일은 건너뜀 */ }
    }
  };
  var queries = [
    "title contains '" + q + "' and mimeType = 'application/pdf' and trashed = false",
    "name contains '" + q + "' and mimeType = 'application/pdf' and trashed = false"
  ];
  for (var i = 0; i < queries.length && !best; i++) {
    try { consider(DriveApp.searchFiles(queries[i]), 200); } catch (e) { /* 다음 문법으로 */ }
  }
  if (!best) { try { consider(DriveApp.getFilesByType('application/pdf'), 500); } catch (e2) { /* 무시 */ } }
  return best;
}

function getSs_() {

function tab_(ss, name, cols) {
  var sh = ss.getSheetByName(name) || ss.insertSheet(name);
  if (sh.getLastRow() === 0) {
    sh.appendRow(cols);
    sh.setFrozenRows(1);
    if (name === JOIN_SHEET) { try { sh.setColumnWidth(9, 420); sh.setColumnWidth(10, 260); } catch (e) { /* 무시 */ } }
  }
  return sh;
}

function headRow_(sh) {
  var w = Math.max(1, sh.getLastColumn());
  return sh.getRange(1, 1, 1, w).getValues()[0].map(function (v) { return String(v == null ? '' : v).trim(); }).filter(function (v, i, arr) {
    // 뒤쪽의 빈 제목은 버림(가운데 빈 칸은 자리 유지)
    for (var j = i; j < arr.length; j++) if (arr[j]) return true;
    return false;
  });
}

function sheetUrl_(ss, sh) {
  var gid = 0;
  try { gid = sh.getSheetId(); } catch (e) { gid = 0; }
  return 'https://docs.google.com/spreadsheets/d/' + ss.getId() + '/edit#gid=' + gid;
}

function logError_(err, raw) {
  try {
    var ss = getSs_();
    var log = ss.getSheetByName('오류로그') || ss.insertSheet('오류로그');
    log.appendRow([new Date(), String(err && err.message ? err.message : err), String(raw || '').slice(0, 1500)]);
  } catch (e2) { /* 무시 */ }
}

function str_(v) { return String(v == null ? '' : v).trim(); }
function lower_(v) { return str_(v).toLowerCase(); }
function stamp_(d) { return Utilities.formatDate(d, TZ, 'yyyy-MM-dd HH:mm'); }
function kb_(n) { return n >= 1048576 ? (n / 1048576).toFixed(1) + 'MB' : Math.max(1, Math.round(n / 1024)) + 'KB'; }
function errText_(err) { return String(err && err.message ? err.message : err); }
function cleanName_(v) {
  var s = String(v == null ? '' : v).replace(/[\u200B-\u200D\uFEFF]/g, '').replace(/[<>]/g, '').replace(/\s+/g, ' ').trim();
  s = s.replace(/\s*님$/, '').trim();
  return s.length > 40 ? s.slice(0, 40) : s;
}
function esc_(s) {
  return String(s == null ? '' : s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
}

     → 처음 한 번 "권한 검토" 창: 계정 선택 → 고급 → (안전하지 않은 페이지로) 이동 → 허용
     → 아래 실행 로그에 점검 결과가 나옵니다.
        · 접수 시트: AI윤리전문가 1기 접수명단 (1H-…)
        · '접수' 탭 N건, '위원 신청' 탭 N건 (열 제목 확인 · 빠진 열 추가)
        · 실행 계정: skkc.office@gmail.com · 관리자 알림 받는 주소: kaiec.korea@gmail.com, skkc.office@gmail.com
        · 오늘 남은 메일 발송 가능 수: N통
        · 지원자 자동 메일: 꺼짐 (지원자에게는 아무 메일도 가지 않고 관리자 알림만 갑니다. …)
  4. 배포: [배포 > 배포 관리] → 연필(수정) → 버전: "새 버전" → [배포]
     ※ '새 배포'를 누르면 주소(/exec)가 바뀌어 사이트가 옛 주소로 계속 보내게 되므로 반드시 '배포 관리 > 새 버전' 입니다.
  5. 확인: https://kaiec.kr/join/ 에서 테스트 제출 → 시트 '위원 신청' 탭에 줄 추가 + 새 제목의 관리자 알림 확인 → 테스트 줄 삭제
     (편집기에서 testJoin 을 실행해도 같은 경로를 점검할 수 있습니다. 2분 안에 다시 실행하면 duplicate 로 기록하지 않습니다)


[3] 관리자 알림 예시 

  제목: [한국AI윤리위원회(KAIEC) AI 윤리 캠페인위원 지원] 홍길동
  본문: 표 + "지원자에게 자동 메일은 가지 않았습니다. 검토 후 지원자 메일로 캠페인위원 참여 제안(활동 안내서)과 활동 안내를
        직접 보내 주십시오." + [위원 신청 탭 열기]
  운영위원 · 전문위원 · 공식 파트너는 "검토 후 3~5일 안에 지원자 메일로 결과와 활동 안내를 보내 주십시오" 로 나옵니다.


[4] 안전장치

  · 같은 이메일 + 같은 참여 구분이 2분 안에 두 번 들어오면(새로고침 등) 한 번만 기록합니다.
  · 양성과정 신청도 같은 사람이 2분 안에 두 번 들어오면 한 번만 기록합니다.
  · 관리자 알림이 실패해도 시트 기록은 남고, 오류는 '오류로그' 탭에 남습니다.
  · 지메일 계정의 Apps Script 발송 한도는 하루 100통(수신자 기준)입니다. 관리자 알림은 신청 1건당 2통(NOTIFY 2주소)이므로
    하루 50건까지 알림이 갑니다. 한도를 넘으면 알림만 빠지고 시트 기록은 남습니다.


[5] 지원자 자동 메일을 켤 때 (나중에)

  켜면 접수 즉시 지원자에게 아래 메일이 나갑니다.
   · AI 윤리 캠페인위원 → "[공식 위원 참여 제안] 한국AI윤리위원회(KAIEC) AI윤리 캠페인위원으로 함께해 주세요"
                          (오준호 사무총장 제안 메일, 15_캠페인위원_제안메일_자동발송 v1.0.0 과 같은 문안, 활동 안내서 PDF 첨부)
   · 운영위원 · 전문위원 · 공식 파트너 → "[한국AI윤리위원회] ○○ 참여 신청이 접수되었습니다" (검토 일정 · 다음 절차 3단계)
   · 양성과정 신청자 → 없음(결제 후 평가 운영 시트가 계정 · 학습자료 메일을 보냄)
  발신 '한국AI윤리위원회 <contact@kaiec.kr>'(실행 계정 지메일의 '다른 주소에서 메일 보내기' 별칭. 없으면 kaiec.korea@gmail.com,
  그것도 없으면 실행 계정 주소), 답장 contact@kaiec.kr.

  순서
   1. 활동 안내서 PDF "한국AI윤리위원회(KAIEC) AI윤리 캠페인위원 활동 안내서.pdf"(관리자 자료 폴더의 활동 안내서 PDF)를 skkc.office 구글 드라이브에 업로드
      (파일 이름에 '캠페인위원' 이 들어 있으면 자동으로 찾습니다)
   2. Code.gs 맨 위 MAIL 블록에서  APPLICANT_ON: false  →  APPLICANT_ON: true  → 저장
   3. checkSetup 실행 → "발신 주소: contact@kaiec.kr (공식 메일 별칭 등록됨)", "활동 안내서 PDF: 정상" 확인
   4. previewMails 실행 → kaiec.korea@gmail.com 으로 견본 3통(제안 메일 · 접수 확인 메일 · 관리자 알림)이 갑니다. 모양 · 첨부 · 발신 주소 확인
   5. [배포 관리 > 연필 > 새 버전 > 배포]
   6. 본인 주소로 /join/ 테스트 제출 → 제안 메일 수신 확인 (test@kaiec.kr 로 제출하면 지원자 메일은 건너뜁니다)

  켰을 때의 안전장치
   · 같은 이메일에 제안 메일은 한 번만(시트 '지원자 메일' 칸의 "제안 메일 발송" 기록으로 판단)
   · test@kaiec.kr, no-reply, example.com 주소에는 보내지 않음(MAIL.SKIP_EMAIL)
   · 남은 발송 한도가 10통 이하면 지원자 메일은 보류(시트에 "보류: …")하고 관리자 알림은 계속
   · 지원자 메일이 실패해도 시트 기록과 관리자 알림은 진행, '지원자 메일' 칸에 "실패: …", '오류로그' 탭에 원인
   · 신청 1건 = 지원자 1통 + 관리자 2통 = 3통 → 무료 지메일 계정(하루 100통)이면 하루 30건 안팎
  MAIL.CAMPAIGN_PROPOSAL: false 로 두면 캠페인위원에게도 제안 메일 대신 접수 확인 메일만 갑니다.


[6] 설정 항목 (Code.gs 맨 위. 바꾼 뒤 저장 → 배포 관리 > 새 버전)

  NOTIFY                        → 관리자 알림 받을 주소(쉼표로 추가)
  MAIL.APPLICANT_ON             → 지원자 자동 메일 (false = 안 보냄, 현재)
  MAIL.CAMPAIGN_PROPOSAL        → 캠페인위원 제안 메일 여부(켰을 때만 의미)
  MAIL.GUIDE_FILE_ID            → 안내서 PDF 를 드라이브 파일 ID 로 고정(이름 검색 대신)
  MAIL.SKIP_EMAIL · QUOTA_RESERVE → 제외 주소 규칙 · 한도 보류 기준
  제안 메일 혜택 문안은 BENEFITS 배열(5항목), 접수 확인 메일 절차는 ackMail_ 의 steps 에 있습니다.


[7] 캠페인위원 제안메일 자동 발송(v1.0.0, 구글 폼용)과의 관계

  · v1.0.0 은 구글 폼 응답에 제안 메일을 보내는 별도 프로젝트로, 설치하지 않았습니다.
  · 6차의 지원자 자동 메일을 켜면 홈페이지 신청서(/join/)로 들어온 캠페인위원에게 같은 문안이 나갑니다. 지금은 둘 다 꺼진 상태이므로
    캠페인위원 제안은 관리자가 알림을 보고 직접 보냅니다.


[8] 문제가 생기면

  · "권한" 오류 → checkSetup 을 편집기에서 다시 실행해 권한을 허용합니다.
  · 사이트 제출 후 시트에 아무것도 없다 → 배포가 '새 배포'로 되어 주소가 바뀌었는지 [배포 관리]에서 확인(사이트는
    build.py 의 SHEET_WEBHOOK 주소로 보냅니다). 편집기에서 testJoin 을 실행하면 시트 기록 · 관리자 알림 경로를 바로 점검할 수 있습니다.
  · 옛 4차 코드로 되돌리려면 이 파일의 이전 판(git 기록, 2026.09.13)에서 꺼내 붙여넣고 새 버전 배포.
